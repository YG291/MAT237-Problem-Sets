\begin{answer}{600pt}{0pt} % Do not change the height of this box
    \textbf{(3c)}

    \begin{proof}
        \setlength{\parskip}{1em}

        Let $S = \{(x, y) \in \R^k \times \R^{n - k} : y = F(x)\}$.

        Since $S$ is the graph of $F$, and $F$ is $C^1$, we have by the tangent space of graphs theorem:
        \[
            T_pS = \{(h, dF_a(h)) : h \in \R^k\}
        \]

        Let $(x, y) \in \R^k \times \R^{n - k}$, then:
        \begin{align*}
            (x, y) \in \Ker(dG_p) & \iff (x, y) \in \Null(DG(p)) \\
            & \iff DG(p) \mat{x \\ y} = \mat{-DF(a) & I_{n - k}} \mat{x \\ y} = 0 \\
            & \iff -DF(a)x + I_{n - k}y = y - DF(a)x = 0 \\
            & \iff y = DF(a)x = dF_a(x) \\
            & \iff (x, y) = (x, dF_a(x)) \\
            & \iff (x, y) \in T_pS
        \end{align*}

        Therefore $T_pS = \Ker(dG_p)$.
    \end{proof}
\end{answer}
