%%% An empty ruled box of fixed height, for handwriting.
%%% Regions are matched in Gradescope, so do not change a height
%%% without re-matching the outline.
\declarecommand{\ruledbox}[1]{%
	\par\nobreak\noindent
	\begingroup
	\setlength{\fboxrule}{0.5pt}%
	\setlength{\fboxsep}{6pt}%
	\fcolorbox{Gray}{white}{%
		\parbox[c][#1][t]{\dimexpr\textwidth-2\fboxsep-2\fboxrule\relax}{%
			\rule{\linewidth}{0pt}% forces the box to full width; an empty parbox has zero width
		}%
	}%
	\endgroup
	\par
}
